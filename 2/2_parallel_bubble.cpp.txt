#include <iostream>
#include <omp.h>
#include <algorithm>

using namespace std;

// ---------------- SEQUENTIAL BUBBLE SORT ----------------
void bubble(int array[], int n) {
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (array[j] > array[j + 1])
                swap(array[j], array[j + 1]);
        }
    }
}

// ---------------- PARALLEL BUBBLE SORT (ODD-EVEN) ----------------
void pBubble(int array[], int n) {

    for (int i = 0; i < n; i++) {

        // EVEN PHASE
        #pragma omp parallel for
        for (int j = 0; j < n - 1; j += 2) {
            if (array[j] > array[j + 1])
                swap(array[j], array[j + 1]);
        }

        // ODD PHASE
        #pragma omp parallel for
        for (int j = 1; j < n - 1; j += 2) {
            if (array[j] > array[j + 1])
                swap(array[j], array[j + 1]);
        }
    }
}

// ---------------- PRINT ----------------
void printArray(int arr[], int n) {
    for (int i = 0; i < n; i++)
        cout << arr[i] << " ";
    cout << "\n";
}

// ---------------- MAIN ----------------
int main() {

    int n = 10;
    int arr[n];

    double start_time, end_time;

    // Reverse array
    for (int i = 0, j = n; i < n; i++, j--)
        arr[i] = j;

    // Sequential
    start_time = omp_get_wtime();
    bubble(arr, n);
    end_time = omp_get_wtime();

    cout << "Sequential Bubble Sort: " << end_time - start_time << " sec\n";
    printArray(arr, n);

    // Reset
    for (int i = 0, j = n; i < n; i++, j--)
        arr[i] = j;

    // Parallel
    start_time = omp_get_wtime();
    pBubble(arr, n);
    end_time = omp_get_wtime();

    cout << "Parallel Bubble Sort: " << end_time - start_time << " sec\n";
    printArray(arr, n);

    return 0;
}
